accentCorner.CornerRadius = UDim.new(0, 16)

-- ทำให้ทุกอย่างในการ์ดใสตอนเริ่ม เพื่อ fade เข้าไปพร้อมกัน
local function setTransparency(obj, t)
    if obj:IsA("TextLabel") then
        obj.TextTransparency = t
    elseif obj:IsA("Frame") and obj.Name ~= "AccentBar" and obj.Name ~= "IconHolder" then
        obj.BackgroundTransparency = t
    end
end

for _, obj in ipairs({title, message}) do
    obj.TextTransparency = 1
end
accentBar.BackgroundTransparency = 1
iconHolder.BackgroundTransparency = 1
exclaim.TextTransparency = 1
pulseStroke.Transparency = 1

-- ========== Animation: เข้า ==========
local cardOpenTween = TweenService:Create(card, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
    Size = UDim2.fromScale(0.86, 0.24),
    BackgroundTransparency = 0,
})

local backdropTween = TweenService:Create(backdrop, TweenInfo.new(0.3), {
    BackgroundTransparency = 0.55,
})

backdropTween:Play()
cardOpenTween:Play()

task.delay(0.25, function()
    TweenService:Create(accentBar, TweenInfo.new(0.3), {BackgroundTransparency = 0}):Play()
    TweenService:Create(iconHolder, TweenInfo.new(0.3), {BackgroundTransparency = 0.85}):Play()
    TweenService:Create(exclaim, TweenInfo.new(0.3), {TextTransparency = 0}):Play()
    TweenService:Create(title, TweenInfo.new(0.3), {TextTransparency = 0}):Play()
    TweenService:Create(message, TweenInfo.new(0.35), {TextTransparency = 0}):Play()
    TweenService:Create(pulseStroke, TweenInfo.new(0.3), {Transparency = 0.2}):Play()
end)

-- ========== Pulse Ring Loop (สัญญาณเตือนต่อเนื่อง) ==========
task.spawn(function()
    while screenGui.Parent do
        pulseRing.Size = UDim2.fromScale(1, 1)
        pulseStroke.Transparency = 0.2
        local grow = TweenService:Create(pulseRing, TweenInfo.new(1.2, Enum.EasingStyle.Sine), {
            Size = UDim2.fromScale(1.6, 1.6),
        })
        local fade = TweenService:Create(pulseStroke, TweenInfo.new(1.2, Enum.EasingStyle.Sine), {
            Transparency = 1,
        })
        grow:Play()
        fade:Play()
        task.wait(1.2)
    end
end)

-- ========== ปุ่มปิด ==========
local closeBtn = Instance.new("TextButton")
closeBtn.Name = "CloseButton"
closeBtn.AnchorPoint = Vector2.new(1, 0)
closeBtn.Position = UDim2.new(1, -12, 0, 12)
closeBtn.Size = UDim2.fromOffset(24, 24)
closeBtn.BackgroundColor3 = Color3.fromRGB(39, 39, 42)
closeBtn.BackgroundTransparency = 0.2
closeBtn.BorderSizePixel = 0
closeBtn.Font = Enum.Font.GothamBold
closeBtn.Text = "×"
closeBtn.TextColor3 = Color3.fromRGB(228, 228, 231)
closeBtn.TextSize = 16
closeBtn.AutoButtonColor = false
closeBtn.ZIndex = 4
closeBtn.Parent = card

local closeCorner = Instance.new("UICorner")
closeCorner.CornerRadius = UDim.new(1, 0)
closeCorner.Parent = closeBtn

closeBtn.MouseEnter:Connect(function()
    TweenService:Create(closeBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(63, 63, 70)}):Play()
end)
closeBtn.MouseLeave:Connect(function()
    TweenService:Create(closeBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(39, 39, 42)}):Play()
end)

local function closeUI()
    local closeTween = TweenService:Create(card, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
        Size = UDim2.new(0.86, 0, 0, 0),
        BackgroundTransparency = 1,
    })
    local backdropClose = TweenService:Create(backdrop, TweenInfo.new(0.3), {
        BackgroundTransparency = 1,
    })
    closeTween:Play()
    backdropClose:Play()
    closeTween.Completed:Connect(function()
        screenGui:Destroy()
    end)
end

closeBtn.MouseButton1Click:Connect(closeUI)

-- ปิดอัตโนมัติหลัง 8 วินาที (ปรับ/ลบได้ตามต้องการ)
task.delay(20, function()
    if screenGui.Parent then
        closeUI()
    end
end)
